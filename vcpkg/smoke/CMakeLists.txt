cmake_minimum_required(VERSION 3.21)
project(cef_package_smoke LANGUAGES CXX)
find_package(cef 152.0.6 EXACT CONFIG REQUIRED)
if(NOT CEF_ENGINE_LINKAGE STREQUAL "SHARED" OR NOT CEF_WRAPPER_LINKAGE STREQUAL "STATIC")
    message(FATAL_ERROR "This is the explicitly hybrid baseline test, not a static-engine test.")
endif()
add_executable(cef_smoke smoke.cc)
target_link_libraries(cef_smoke PRIVATE CEF::wrapper)
if(MSVC)
    target_compile_options(cef_smoke PRIVATE /utf-8 /W4 /wd4100)
    target_compile_definitions(cef_smoke PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
endif()
cef_deploy_runtime(cef_smoke)
