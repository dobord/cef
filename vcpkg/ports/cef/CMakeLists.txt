cmake_minimum_required(VERSION 3.21)
project(cef_vcpkg VERSION 152.0.6 LANGUAGES C CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
set(CEF_ROOT "${CMAKE_CURRENT_SOURCE_DIR}")
set(USE_SANDBOX OFF CACHE BOOL "The package does not supply the Windows bootstrap/sandbox integration" FORCE)
set(USE_ATL OFF CACHE BOOL "Do not require ATL to build the wrapper" FORCE)
set(USE_INSTALLER OFF CACHE BOOL "No online runtime installer" FORCE)
set(api_version 15200)
set(CEF_RUNTIME_LIBRARY_FLAG "" CACHE STRING "Honor the vcpkg-selected CMAKE_MSVC_RUNTIME_LIBRARY" FORCE)
set(CEF_DEBUG_INFO_FLAG "/Z7" CACHE STRING "Embed debug info in static objects" FORCE)
# Upstream clears these on Windows/Ninja. Preserve the vcpkg toolchain's flags.
foreach(flag CMAKE_CXX_FLAGS CMAKE_CXX_FLAGS_DEBUG CMAKE_CXX_FLAGS_RELEASE)
    set(saved_${flag} "${${flag}}")
endforeach()
list(APPEND CMAKE_MODULE_PATH "${CEF_ROOT}/cmake")
find_package(CEF REQUIRED MODULE)
foreach(flag CMAKE_CXX_FLAGS CMAKE_CXX_FLAGS_DEBUG CMAKE_CXX_FLAGS_RELEASE)
    set(${flag} "${saved_${flag}}")
endforeach()
add_subdirectory(libcef_dll)
set_target_properties(libcef_dll_wrapper PROPERTIES EXPORT_NAME wrapper POSITION_INDEPENDENT_CODE ON)
target_compile_features(libcef_dll_wrapper PUBLIC cxx_std_20)
target_compile_definitions(libcef_dll_wrapper INTERFACE CEF_API_VERSION=15200)
target_include_directories(libcef_dll_wrapper PUBLIC
    $<BUILD_INTERFACE:${CEF_ROOT}>
    $<INSTALL_INTERFACE:include/cef>
)
add_library(CEF::runtime SHARED IMPORTED GLOBAL)
if(WIN32)
    set_target_properties(CEF::runtime PROPERTIES IMPORTED_LOCATION "${CEF_ROOT}/Release/libcef.dll" IMPORTED_IMPLIB "${CEF_ROOT}/Release/libcef.lib")
    target_link_libraries(libcef_dll_wrapper INTERFACE CEF::runtime ${CEF_STANDARD_LIBS})
    target_link_options(libcef_dll_wrapper INTERFACE /DELAYLOAD:libcef.dll /STACK:8388608)
else()
    find_package(Threads REQUIRED)
    find_package(X11 REQUIRED)
    set_target_properties(CEF::runtime PROPERTIES IMPORTED_LOCATION "${CEF_ROOT}/Release/libcef.so" IMPORTED_SONAME libcef.so)
    target_link_libraries(libcef_dll_wrapper INTERFACE CEF::runtime Threads::Threads X11::X11 ${CMAKE_DL_LIBS})
endif()
install(TARGETS libcef_dll_wrapper EXPORT cef-targets ARCHIVE DESTINATION "${CMAKE_INSTALL_LIBDIR}")
install(DIRECTORY include/ DESTINATION include/cef/include FILES_MATCHING PATTERN "*.h")
# Runtime binaries/resources are an inseparable deployment bundle, not static libraries.
foreach(runtime_file IN LISTS CEF_BINARY_FILES)
    if(NOT EXISTS "${CEF_ROOT}/Release/${runtime_file}")
        message(FATAL_ERROR "Missing required runtime file: ${runtime_file}")
    endif()
    install(PROGRAMS "${CEF_ROOT}/Release/${runtime_file}" DESTINATION tools/cef CONFIGURATIONS Release RelWithDebInfo MinSizeRel)
endforeach()
if(WIN32)
    install(FILES "${CEF_ROOT}/Release/libcef.lib" DESTINATION tools/cef CONFIGURATIONS Release RelWithDebInfo MinSizeRel)
endif()
install(DIRECTORY Resources/ DESTINATION tools/cef CONFIGURATIONS Release RelWithDebInfo MinSizeRel)
install(FILES LICENSE.txt README.txt DESTINATION tools/cef CONFIGURATIONS Release RelWithDebInfo MinSizeRel)
configure_package_config_file(cef-config.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/cef-config.cmake" INSTALL_DESTINATION lib/cmake/cef)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/cef-config-version.cmake" VERSION ${PROJECT_VERSION} COMPATIBILITY ExactVersion)
install(EXPORT cef-targets FILE cef-targets.cmake NAMESPACE CEF:: DESTINATION lib/cmake/cef)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/cef-config.cmake" "${CMAKE_CURRENT_BINARY_DIR}/cef-config-version.cmake" cef-deploy.cmake DESTINATION lib/cmake/cef)
